# frozen_string_literal: true

require_relative "test_helper"
require "pstore"
require "tmpdir"

class TestPStoreLoggerStringFileId < Minitest::Test
  def setup
    @temp_dir = Dir.mktmpdir
    @logger = PStoreLogger.new(storage_path: @temp_dir, tag: "Customer")
  end

  def teardown
    FileUtils.remove_entry(@temp_dir)
  end

  def test_documented_string_file_id_saves_unique_files_with_their_payloads
    first_data = { id: "user_123", event: "created" }
    second_data = { id: "user_123", event: "updated" }

    first_path = @logger.save(first_data, file_id: "user_123")
    second_path = @logger.save(second_data, file_id: "user_123")

    assert_equal "user_123.pstore", File.basename(first_path)
    assert_equal "user_123_1.pstore", File.basename(second_path)
    assert_equal first_data, stored_data(first_path)
    assert_equal second_data, stored_data(second_path)
  end

  private

    def stored_data(path)
      PStore.new(path).transaction(true) { |store| store[:data] }
    end
end
