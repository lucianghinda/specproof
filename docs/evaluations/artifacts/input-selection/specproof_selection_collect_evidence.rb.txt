def archive_workspace(source, target)
  FileUtils.mkdir_p(target)
  files = Dir.glob(File.join(source, "**/*"), File::FNM_DOTMATCH).select { |path| File.file?(path) }.sort
  members = files.to_h { |path| [path.delete_prefix(source + "/"), Digest::SHA256.file(path).hexdigest] }
  archive = File.join(target, "workspace.tar.gz")
  Zlib::GzipWriter.open(archive) do |gzip|
    Gem::Package::TarWriter.new(gzip) do |tar|
      files.each do |path|
        relative = path.delete_prefix(source + "/")
        data = File.binread(path)
        tar.add_file_simple(relative, 0o644, data.bytesize) { |entry| entry.write(data) }
      end
    end
  end
  File.write(File.join(target, "manifest.json"), JSON.pretty_generate(archive_sha256: Digest::SHA256.file(archive).hexdigest, files: members) + "\n")
  result = File.join(source, "RESULT.md")
  FileUtils.cp(result, target) if File.file?(result)
end

end
