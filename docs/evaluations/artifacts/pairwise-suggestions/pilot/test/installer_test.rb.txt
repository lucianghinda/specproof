# frozen_string_literal: true

require_relative "test_helper"

class InstallerTest < Minitest::Test
  def setup
    @root = Dir.mktmpdir
    @home = Dir.mktmpdir
    @installer = AllInRuby::Installer.new(root: @root, home: @home, env: {})
  end

  def teardown
    FileUtils.remove_entry(@root)
    FileUtils.remove_entry(@home)
  end

  def test_instruction_is_installed_only_with_a_complete_marked_block
    assert AllInRuby::Instruction.installed?(AllInRuby::Instruction.block)
    refute AllInRuby::Instruction.installed?(AllInRuby::Instruction::BEGIN_MARKER)
  end

  def test_instruction_remove_returns_empty_content_for_the_entire_block
    assert_empty AllInRuby::Instruction.remove(AllInRuby::Instruction.block)
  end

  def test_instruction_remove_preserves_unrelated_content
    content = "Before\n\n#{AllInRuby::Instruction.block}After\n"

    assert_equal "Before\n\nAfter\n", AllInRuby::Instruction.remove(content)
  end

  def test_instruction_remove_does_not_cross_a_second_begin_marker
    dangling = "#{AllInRuby::Instruction::BEGIN_MARKER}\nKeep this text.\n\n"

    assert_equal dangling, AllInRuby::Instruction.remove(dangling + AllInRuby::Instruction.block)
  end

  def test_install_creates_local_files_for_both_agents
    results = @installer.install(agents: %w[claude codex], scope: "local")

    assert_equal %i[created created], results.map(&:status)
    assert_includes File.read(File.join(@root, "CLAUDE.md")), "write it in Ruby"
    assert_includes File.read(File.join(@root, "AGENTS.md")), "write it in Ruby"
  end

  def test_install_appends_without_touching_existing_content
    path = File.join(@root, "CLAUDE.md")
    existing = "# My project\n\nAlways use tabs.\n"
    File.write(path, existing)

    results = @installer.install(agents: ["claude"], scope: "local")

    assert_equal [:updated], results.map(&:status)
    content = File.read(path)
    assert content.start_with?(existing), "existing content must be preserved verbatim"
    assert_includes content, AllInRuby::Instruction::BEGIN_MARKER
  end

  def test_install_is_idempotent
    @installer.install(agents: ["claude"], scope: "local")
    before = File.read(File.join(@root, "CLAUDE.md"))

    results = @installer.install(agents: ["claude"], scope: "local")

    assert_equal [:already_installed], results.map(&:status)
    assert_equal before, File.read(File.join(@root, "CLAUDE.md"))
  end

  def test_install_global_creates_files_under_dot_directories
    results = @installer.install(agents: %w[claude codex], scope: "global")

    assert_equal %i[created created], results.map(&:status)
    assert File.exist?(File.join(@home, ".claude", "CLAUDE.md"))
    assert File.exist?(File.join(@home, ".codex", "AGENTS.md"))
  end

  def test_uninstall_removes_only_our_block
    path = File.join(@root, "CLAUDE.md")
    File.write(path, "# My project\n\nAlways use tabs.\n")
    @installer.install(agents: ["claude"], scope: "local")

    results = @installer.uninstall(agents: ["claude"], scope: "local")

    assert_equal [:removed], results.map(&:status)
    content = File.read(path)
    assert_includes content, "Always use tabs."
    refute_includes content, AllInRuby::Instruction::BEGIN_MARKER
    refute_includes content, "write it in Ruby"
  end

  def test_uninstall_when_not_installed
    results = @installer.uninstall(agents: ["claude"], scope: "local")

    assert_equal [:not_installed], results.map(&:status)
    refute File.exist?(File.join(@root, "CLAUDE.md"))
  end

  def test_status_reports_per_agent
    @installer.install(agents: ["claude"], scope: "local")

    results = @installer.status(agents: %w[claude codex], scope: "local")

    assert_equal %i[installed not_installed], results.map(&:status)
  end

  def test_codex_global_respects_codex_home_env
    codex_home = File.join(@home, "custom-codex")
    installer = AllInRuby::Installer.new(root: @root, home: @home, env: { "CODEX_HOME" => codex_home })

    results = installer.install(agents: ["codex"], scope: "global")

    assert_equal [:created], results.map(&:status)
    assert File.exist?(File.join(codex_home, "AGENTS.md"))
    refute File.exist?(File.join(@home, ".codex", "AGENTS.md"))
  end

  def test_codex_global_ignores_empty_codex_home_env
    installer = AllInRuby::Installer.new(root: @root, home: @home, env: { "CODEX_HOME" => "" })

    installer.install(agents: ["codex"], scope: "global")

    assert File.exist?(File.join(@home, ".codex", "AGENTS.md"))
  end

  def test_opencode_local_shares_agents_md_with_codex
    results = @installer.install(agents: %w[codex opencode cursor], scope: "local")

    assert_equal %i[created already_installed already_installed], results.map(&:status)
    assert_equal [File.join(@root, "AGENTS.md")] * 3, results.map(&:path)
  end

  def test_opencode_global_creates_file_under_config_directory
    results = @installer.install(agents: ["opencode"], scope: "global")

    assert_equal [:created], results.map(&:status)
    assert_includes File.read(File.join(@home, ".config", "opencode", "AGENTS.md")), "write it in Ruby"
  end

  def test_cursor_local_installs_into_agents_md
    results = @installer.install(agents: ["cursor"], scope: "local")

    assert_equal [:created], results.map(&:status)
    assert_includes File.read(File.join(@root, "AGENTS.md")), "write it in Ruby"
  end

  def test_cursor_global_is_manual_and_writes_nothing
    %i[install uninstall status].each do |command|
      results = @installer.public_send(command, agents: ["cursor"], scope: "global")

      assert_equal [:manual], results.map(&:status)
      assert_nil results.first.path
    end
    assert_empty Dir.children(@home)
  end

  def test_install_appends_with_blank_line_when_file_lacks_trailing_newline
    path = File.join(@root, "AGENTS.md")
    File.write(path, "# Notes")

    @installer.install(agents: ["codex"], scope: "local")

    assert_includes File.read(path), "# Notes\n\n#{AllInRuby::Instruction::BEGIN_MARKER}"
  end

  def test_install_without_dry_run_carries_no_diff
    results = @installer.install(agents: ["claude"], scope: "local")

    assert_nil results.first.diff
  end

  def test_dry_run_install_writes_nothing
    results = @installer.install(agents: %w[claude codex], scope: "local", dry_run: true)

    assert_equal %i[created created], results.map(&:status)
    assert_empty Dir.children(@root)
  end

  def test_dry_run_install_diff_for_new_file
    path = File.join(@root, "CLAUDE.md")

    results = @installer.install(agents: ["claude"], scope: "local", dry_run: true)

    diff = results.first.diff
    assert_includes diff, "--- /dev/null\n+++ #{path}\n@@ -0,0 +1,"
    assert_includes diff, "+#{AllInRuby::Instruction::BEGIN_MARKER}"
    assert_includes diff, "+## Scripts"
  end

  def test_dry_run_install_diff_locates_append_after_existing_content
    path = File.join(@root, "CLAUDE.md")
    File.write(path, "# My project\n\nAlways use tabs.\n")

    results = @installer.install(agents: ["claude"], scope: "local", dry_run: true)

    diff = results.first.diff
    assert_includes diff, "--- #{path}\n+++ #{path}\n@@ -1,3 +1,"
    assert_includes diff, " Always use tabs.\n+\n+#{AllInRuby::Instruction::BEGIN_MARKER}"
    assert_equal "# My project\n\nAlways use tabs.\n", File.read(path)
  end

  def test_dry_run_install_diff_marks_missing_trailing_newline
    File.write(File.join(@root, "AGENTS.md"), "# Notes")

    results = @installer.install(agents: ["codex"], scope: "local", dry_run: true)

    assert_includes results.first.diff, "-# Notes\n\\ No newline at end of file\n+# Notes"
  end

  def test_dry_run_install_simulates_shared_agents_md
    results = @installer.install(agents: %w[codex opencode cursor], scope: "local", dry_run: true)

    assert_equal %i[created already_installed already_installed], results.map(&:status)
    assert_equal 1, results.map(&:diff).compact.length
  end

  def test_dry_run_uninstall_shows_removal_without_touching_file
    path = File.join(@root, "CLAUDE.md")
    File.write(path, "# My project\n\nAlways use tabs.\n")
    @installer.install(agents: ["claude"], scope: "local")
    before = File.read(path)

    results = @installer.uninstall(agents: ["claude"], scope: "local", dry_run: true)

    assert_equal [:removed], results.map(&:status)
    assert_includes results.first.diff, "-#{AllInRuby::Instruction::BEGIN_MARKER}"
    assert_includes results.first.diff, "-## Scripts"
    assert_equal before, File.read(path)
  end

  # Backups: an existing file is copied before it is modified. The clock is
  # injectable so the timestamped name is deterministic in tests.
  BACKUP_SUFFIX = ".20260802-153000.bak"

  def clocked_installer
    AllInRuby::Installer.new(root: @root, home: @home, env: {},
                             clock: -> { Time.new(2026, 8, 2, 15, 30, 0) })
  end

  def test_install_backs_up_existing_file_before_updating
    path = File.join(@root, "CLAUDE.md")
    original = "# My project\n\nAlways use tabs.\n"
    File.write(path, original)

    results = clocked_installer.install(agents: ["claude"], scope: "local")

    backup = "#{path}#{BACKUP_SUFFIX}"
    assert_equal backup, results.first.backup
    assert_equal original, File.read(backup)
    assert_includes File.read(path), "write it in Ruby"
  end

  def test_install_creating_new_file_makes_no_backup
    results = clocked_installer.install(agents: ["claude"], scope: "local")

    assert_nil results.first.backup
    assert_empty Dir.glob(File.join(@root, "*.bak"))
  end

  def test_install_already_installed_makes_no_backup
    @installer.install(agents: ["claude"], scope: "local")

    results = clocked_installer.install(agents: ["claude"], scope: "local")

    assert_equal [:already_installed], results.map(&:status)
    assert_nil results.first.backup
    assert_empty Dir.glob(File.join(@root, "*.bak"))
  end

  def test_uninstall_backs_up_before_removing
    path = File.join(@root, "CLAUDE.md")
    File.write(path, "# My project\n\nAlways use tabs.\n")
    @installer.install(agents: ["claude"], scope: "local")
    before = File.read(path)

    results = clocked_installer.uninstall(agents: ["claude"], scope: "local")

    backup = "#{path}#{BACKUP_SUFFIX}"
    assert_equal backup, results.first.backup
    assert_equal before, File.read(backup)
    refute_includes File.read(path), "write it in Ruby"
  end

  def test_backup_can_be_disabled
    path = File.join(@root, "CLAUDE.md")
    File.write(path, "# My project\n")

    results = clocked_installer.install(agents: ["claude"], scope: "local", backup: false)

    assert_equal [:updated], results.map(&:status)
    assert_nil results.first.backup
    assert_empty Dir.glob(File.join(@root, "*.bak"))
    assert_includes File.read(path), "write it in Ruby"
  end

  def test_shared_agents_md_is_backed_up_once
    File.write(File.join(@root, "AGENTS.md"), "# Notes\n")

    results = clocked_installer.install(agents: %w[codex opencode cursor], scope: "local")

    assert_equal %i[updated already_installed already_installed], results.map(&:status)
    assert_equal 1, results.map(&:backup).compact.length
    assert_equal 1, Dir.glob(File.join(@root, "*.bak")).length
  end

  def test_dry_run_names_the_backup_without_creating_it
    path = File.join(@root, "CLAUDE.md")
    File.write(path, "# My project\n")

    results = clocked_installer.install(agents: ["claude"], scope: "local", dry_run: true)

    backup = "#{path}#{BACKUP_SUFFIX}"
    assert_equal backup, results.first.backup
    refute File.exist?(backup)
    assert_equal "# My project\n", File.read(path)
  end
end
