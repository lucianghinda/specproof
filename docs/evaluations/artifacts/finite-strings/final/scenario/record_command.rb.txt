require "json"
require "open3"
require "time"

root = File.expand_path("..", __dir__)
scenario = __dir__
config = JSON.parse(File.read(File.join(root, "environment.json")))
log_path, *argv = ARGV
abort "usage: record_command.rb LOG_PATH COMMAND [ARG ...]" unless log_path && !argv.empty?

started_at = Time.now.utc.iso8601(6)
stdout, stderr, status = Open3.capture3(config, *argv, chdir: scenario, unsetenv_others: true)
record = {
  started_at: started_at,
  finished_at: Time.now.utc.iso8601(6),
  cwd: scenario,
  argv: argv,
  env: config,
  exit_status: status.exitstatus,
  success: status.success?,
  stdout: stdout,
  stderr: stderr
}
File.write(log_path, JSON.pretty_generate(record) + "\n")
STDOUT.write(stdout)
STDERR.write(stderr)
exit(status.exitstatus || 1)
